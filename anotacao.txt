type HeaderItem = {

type TestTaskPayload = {
  endpoint?: string;
  metodo?: string;

  tipo_autenticacao?: "nenhuma" | "basic" | "bearer" | "api_key" | null;

  usuario?: string | null;
  senha?: string | null;

  headers?: HeaderItem[];

  timeout_segundos?: number | string;
};

function jsonResponse(data: Record<string, unknown>, status = 200) {
  return NextResponse.json(data, { status });
}

/**
 * POST /api/tarefas/testar
 *
 * Botão "Testar" da tela de tarefas: executa a requisição configurada
 * uma única vez e devolve o resultado, sem persistir nada.
 *
 * Exige usuário autenticado — sem esse check qualquer um poderia usar
 * este endpoint como proxy para bater em URLs arbitrárias a partir do
 * nosso servidor (SSRF).
 */
export async function POST(request: NextRequest) {
  const inicio = Date.now();

  try {
    const supabase = await createClient();

    const {
      data: { user },
      error: authError,
    } = await supabase.auth.getUser();

    if (authError || !user) {
      return jsonResponse(
        {
          success: false,
          error: "Usuário não autenticado.",
        },
        401,
      );
    }

    let body: TestTaskPayload;

    try {
      body = await request.json();
    } catch {
      return jsonResponse(
        {
          success: false,
          error: "JSON inválido.",
        },
        400,
      );
    }

    let resultado;

    try {
      resultado = await executarRequisicaoTarefa({
        endpoint: body.endpoint || "",
        metodo: body.metodo,
        tipo_autenticacao: body.tipo_autenticacao,
        usuario: body.usuario,
        senha: body.senha,
        headers: body.headers,
        timeout_segundos: body.timeout_segundos,
      });
    } catch (error) {
      return jsonResponse(
        {
          success: false,
          error:
            error instanceof Error
              ? error.message
              : "Configuração inválida da tarefa.",
        },
        400,
      );
    }

    /*
     * Retornamos HTTP 200 para o próprio teste mesmo quando o endpoint
     * testado retornou erro/timeout.
     *
     * 200 = o sistema conseguiu executar o teste.
     * success = false = o endpoint testado apresentou erro.
     */
    return jsonResponse({
      success: true,
      sucesso: resultado.sucesso,
      status: resultado.sucesso ? "sucesso" : "erro",
      mensagem: resultado.timeout
        ? "A requisição excedeu o tempo limite."
        : resultado.sucesso
          ? "Endpoint respondeu com sucesso."
          : `Endpoint respondeu HTTP ${resultado.http_status ?? "desconhecido"}.`,

      endpoint: body.endpoint,
      metodo: (body.metodo || "GET").toUpperCase(),

      http_status: resultado.http_status,
      http_status_text: resultado.http_status_text,
      tempo_ms: resultado.tempo_ms,

      resposta: resultado.resposta,
      resposta_truncada: resultado.resposta_truncada,
      headers_resposta: resultado.headers_resposta,

      erro: resultado.erro,
      timeout: resultado.timeout,
    });
  } catch (error) {
    console.error("Erro inesperado ao testar tarefa:", error);

    return jsonResponse(
      {
        success: false,
        status: "erro",
        mensagem: "Erro interno ao testar a tarefa.",
        erro: error instanceof Error ? error.message : "Erro desconhecido.",
        tempo_ms: Date.now() - inicio,
      },
      500,
    );
  }
}

function formatDate(value: string | null) {
  if (!value) return "-";

  try {
    return new Intl.DateTimeFormat("pt-BR", {
      dateStyle: "short",
      timeStyle: "short",
    }).format(new Date(value));
  } catch {
    return value;
  }
}

function getStatusLabel(status: string | null) {
  if (!status) return "Nunca executada";

  const normalized = status.toLowerCase();

  if (
    normalized === "sucesso" ||
    normalized === "success" ||
    normalized === "ok"
  ) {
    return "Sucesso";
  }

  if (
    normalized === "erro" ||
    normalized === "error" ||
    normalized === "failed" ||
    normalized === "falha"
  ) {
    return "Erro";
  }

  return status;
}

function getScheduleLabel(task: Task) {
  if (
    task.tipo_agendamento === "intervalo" ||
    (!task.tipo_agendamento && task.cron_expression)
  ) {
    if (task.intervalo_minutos) {
      return `A cada ${task.intervalo_minutos} min`;
    }

    if (task.cron_expression) {
      return task.cron_expression;
    }

    return "Intervalo";
  }

  switch (task.tipo_agendamento) {
    case "diario":
      return task.horario
        ? `Diário às ${task.horario.substring(0, 5)}`
        : "Diário";

    case "unica":
      return task.data_execucao
        ? `${task.data_execucao} ${task.horario?.substring(0, 5) ?? ""}`
        : "Uma vez";

    case "customizado":
      return `Customizado${
        task.horario ? ` às ${task.horario.substring(0, 5)}` : ""
      }`;

    default:
      return task.cron_expression || "-";
  }
}

/**
 * Gera automaticamente a expressão CRON.
 */
function generateCronExpression(form: TaskForm) {
  switch (form.tipo_agendamento) {
    case "intervalo": {
      const minutos = Number(form.intervalo_minutos);

      if (!minutos || minutos < 1) {
        throw new Error("Informe um intervalo válido em minutos.");
      }

      if (minutos === 1) {
        return "* * * * *";
      }

      return `*/${minutos} * * * *`;
    }

    case "diario": {
      if (!form.horario) {
        throw new Error("Informe o horário da execução.");
      }

      const [hora, minuto] = form.horario.split(":").map(Number);

      if (Number.isNaN(hora) || Number.isNaN(minuto)) {
        throw new Error("Informe um horário válido.");
      }

      return `${minuto} ${hora} * * *`;
    }

    case "unica": {
      if (!form.data_execucao) {
        throw new Error("Informe a data da execução.");
      }

      if (!form.horario) {
        throw new Error("Informe o horário da execução.");
      }

      const [hora, minuto] = form.horario.split(":").map(Number);

      const [, mes, dia] = form.data_execucao.split("-").map(Number);

      if (
        Number.isNaN(hora) ||
        Number.isNaN(minuto) ||
        Number.isNaN(mes) ||
        Number.isNaN(dia)
      ) {
        throw new Error("Informe uma data e horário válidos.");
      }

      return `${minuto} ${hora} ${dia} ${mes} *`;
    }

    case "customizado": {
      if (!form.horario) {
        throw new Error("Informe o horário da execução.");
      }

      const [hora, minuto] = form.horario.split(":").map(Number);

      if (Number.isNaN(hora) || Number.isNaN(minuto)) {
        throw new Error("Informe um horário válido.");
      }

      const dia = form.dia_mes.trim() || "*";
      const mes = form.mes.trim() || "*";

      return `${minuto} ${hora} ${dia} ${mes} *`;
    }

    default:
      throw new Error("Informe o tipo de agendamento.");
  }
}

  /**
   * Testa a tarefa usando os dados atuais do modal.
   *
   * Não salva a tarefa no banco.
   */
  async function testarTarefa() {
    try {
      setTestando(true);
      setError("");
      setSuccess("");

      if (!form.endpoint.trim()) {
        throw new Error("Informe a URL / Endpoint antes de testar.");
      }

      if (!form.metodo) {
        throw new Error("Informe o método HTTP.");
      }

      const timeout = Number(form.timeout_segundos);

      if (!timeout || timeout < 1 || timeout > 300) {
        throw new Error("O timeout deve estar entre 1 e 300 segundos.");
      }

      const headers = form.headers.filter(
        (header) => header.key.trim() && header.value.trim(),
      );

      const response = await fetch("/api/tarefas/testar", {
        method: "POST",

        headers: {
          "Content-Type": "application/json",
        },

        credentials: "include",

        body: JSON.stringify({
          endpoint: form.endpoint.trim(),

          metodo: form.metodo,

          headers,

          tipo_autenticacao:
            form.tipo_autenticacao === "nenhuma"
              ? null
              : form.tipo_autenticacao,

          usuario: form.usuario.trim() || null,

          senha: form.senha || null,

          timeout_segundos: timeout,
        }),
      });

      const data = await response.json();

      if (!response.ok || (!data.success && !data.sucesso)) {
        throw new Error(
          data.error ||
            data.erro ||
            data.message ||
            "A tarefa não pôde ser executada.",
        );
      }

      const httpStatus =
        data.http_status ?? data.status_code ?? data.statusCode ?? null;

      const tempoMs = data.tempo_ms ?? data.tempo ?? data.duration_ms ?? null;

      if (httpStatus !== null && httpStatus >= 200 && httpStatus < 300) {
        setSuccess(
          `Teste executado com sucesso. HTTP ${httpStatus}${
            tempoMs !== null ? ` em ${tempoMs} ms` : ""
          }.`,
        );
      } else if (httpStatus !== null) {
        const mensagemHttp =
          httpStatus === 400
            ? "Requisição inválida."
            : httpStatus === 401
              ? "Não autorizado. Verifique as credenciais."
              : httpStatus === 403
                ? "Acesso proibido."
                : httpStatus === 404
                  ? "Endpoint não encontrado."
                  : httpStatus >= 500
                    ? "O servidor do endpoint apresentou um erro."
                    : "O endpoint respondeu com erro.";

        setError(
          `Endpoint respondeu HTTP ${httpStatus}${
            tempoMs !== null ? ` em ${tempoMs} ms` : ""
          }. ${mensagemHttp}`,
        );
      } else {
        setError(
          data.message ||
            data.mensagem ||
            data.erro ||
            "O endpoint não respondeu corretamente.",
        );
      }
    } catch (err) {
      console.error("Erro ao testar tarefa:", err);

      setError(
        err instanceof Error
          ? err.message
          : "Não foi possível executar o teste da tarefa.",
      );
    } finally {
      setTestando(false);
    }
  }

        throw new Error("O número de tentativas deve estar entre 1 e 20.");

      if (!timeout || timeout < 1 || timeout > 300) {
        throw new Error("O timeout deve estar entre 1 e 300 segundos.");

          : null,

        data_execucao: form.data_execucao || null,

        mes: form.mes ? Number(form.mes) : null,

        ano: form.ano ? Number(form.ano) : null,

        method: editingTask ? "PATCH" : "POST",

        headers: {
          "Content-Type": "application/json",
        },

        credentials: "include",

      });

        throw new Error(data.error || "Não foi possível salvar a tarefa.");

        editingTask

      setError(
        err instanceof Error
          ? err.message
          : "Não foi possível salvar a tarefa.",
      );
    } finally {
      setSaving(false);
    }

        throw new Error(data.error || "Não foi possível excluir a tarefa.");

/\*

 \* Sincronização com o cron-job.org:

 \*

 \* Cada tarefa possui um job correspondente no cron-job.org

 \* (id salvo em tasks.cronjob\_id), que faz uma chamada GET para

 \* /api/cron/executar-tarefas/{id} no horário configurado, autenticada

 \* via "Authorization: Bearer \<CRON\_SECRET>".

 \*

 \* Falhas ao sincronizar com o cron-job.org NUNCA bloqueiam a operação

 \* local (criar/editar/excluir a tarefa) — apenas são logadas e

 \* devolvidas como aviso. Isso evita que uma instabilidade no serviço

 \* externo impeça o uso normal da tela de tarefas.

 \*/

function getBaseUrl(): string | null {

  const baseUrl = process.env.APP\_BASE\_URL;

  if (!baseUrl) {

    console.error(

      "[tarefas] APP\_BASE\_URL não configurada — pulando sincronização com o cron-job.org."

    );

    return null;

  }

  return baseUrl.replace(/**\\/**+$/, "");

}

type TaskParaSincronizar = {

  id: string;

  nome: string;

  ativa: boolean;

  timeout\_segundos: number | null;

  tipo\_agendamento: "intervalo" | "diario" | "unica" | "customizado";

  intervalo\_minutos: number | null;

  horario: string | null;

  data\_execucao: string | null;

  cron\_expression: string | null;

  timezone: string | null;

};

function montarPayloadCronJob(task: TaskParaSincronizar) {

  const baseUrl = getBaseUrl();

  const cronSecret = process.env.CRON\_SECRET;

  if (!baseUrl || !cronSecret) {

    return null;

  }

  const schedule = construirSchedule({

    tipo\_agendamento: task.tipo\_agendamento,

    intervalo\_minutos: task.intervalo\_minutos,

    horario: task.horario,

    data\_execucao: task.data\_execucao,

    cron\_expression: task.cron\_expression,

    timezone: task.timezone,

  });

  return {

    url: \`${baseUrl}/api/cron/executar-tarefas/${task.id}\`,

    enabled: task.ativa,

    title: \`[monitoramento] ${task.nome}\`.slice(0, 255),

    saveResponses: false,

    // O cron-job.org (plano gratuito) encerra a conexão em \~30s

    // esperando resposta; limitamos aqui para não marcar como falha

    // no painel deles mesmo quando a tarefa está OK.

    requestTimeout: Math.min(task.timeout\_segundos || 30, 30),

    requestMethod: CRONJOB\_REQUEST\_METHOD.GET,

    schedule,

    extendedData: {

      headers: {

        Authorization: \`Bearer ${cronSecret}\`,

      },

    },

  };

}

/\*\*

 \* Cria ou atualiza o job no cron-job.org para a tarefa recém-salva, e

 \* persiste o cronjob\_id quando for criação. Nunca lança — devolve uma

 \* mensagem de aviso (ou null) para o chamador incluir na resposta.

 \*/

async function sincronizarCronJob(

  supabase: Awaited\<ReturnType\<typeof createClient>>,

  data: TaskParaSincronizar & { cronjob\_id: number | null }

): Promise\<string | null> {

  try {

    const payload = montarPayloadCronJob(data);

    if (!payload) {

      return "Tarefa salva, mas a sincronização com o cron-job.org está desativada (variáveis de ambiente ausentes).";

    }

    if (data.cronjob\_id) {

      await atualizarCronJob(data.cronjob\_id, payload);

    } else {

      const jobId = await criarCronJob(payload);

      const { error } = await supabase

        .from("tasks")

        .update({ cronjob\_id: jobId })

        .eq("id", data.id);

      if (error) {

        console.error("[tarefas] Erro ao salvar cronjob\_id:", error);

      } else {

        data.cronjob\_id = jobId;

      }

    }

    return null;

  } catch (error) {

    console.error(

      "[tarefas] Erro ao sincronizar job no cron-job.org:",

      error

    );

    return "Tarefa salva, mas não foi possível sincronizar o agendamento no cron-job.org. Tente salvar novamente em instantes.";

  }

}

/\*\*

 \*

 \* Lista todas as tarefas cadastradas.

export async function GET() {

  try {

    const supabase = await createClient();

    const {

      data: { user },

      error: authError,

    } = await supabase.auth.getUser();

    if (authError || !user) {

      console.error(

        "[tarefas] Usuário não autenticado:",

        authError?.message

      );

      return jsonError(

        "Usuário não autenticado.",

        401

      );

    }

    const { data, error } = await supabase

      .from("tasks")

      .select(TASK\_FIELDS)

      .order("created\_at", {

        ascending: false,

      });

    if (error) {

      console.error(

        "[tarefas] Erro ao buscar tarefas:",

        error

      );

      return jsonError(

        "Não foi possível carregar as tarefas.",

        500

      );

    }

    return NextResponse.json({

      success: true,

      data: data ?? [],

    });

  } catch (error) {

    console.error(

      "[tarefas] GET erro inesperado:",

      error

    );

    return jsonError(

      "Erro interno ao carregar as tarefas.",

      500

    );

  }

}

    const headers = Array.isArray(

    )

      ? body.headers.filter(

          (header) =>

            header &&

            header.key?.trim() &&

            header.value?.trim()

        )

      : [];

    const insertData: Record<

        body.ativa ?? true,

      tipo\_agendamento:

        body.tipo\_agendamento ||

        "intervalo",

      intervalo\_minutos:

        body.intervalo\_minutos !==

        undefined

          ? body.intervalo\_minutos

          : null,

      horario:

        body.horario || null,

      data\_execucao:

        body.data\_execucao || null,

      dia\_mes:

        body.dia\_mes !== undefined

          ? body.dia\_mes

          : null,

      mes:

        body.mes !== undefined

          ? body.mes

          : null,

      ano:

        body.ano !== undefined

          ? body.ano

          : null,

      timezone:

        body.timezone ||

        "America/Sao\_Paulo",

      tipo\_autenticacao:

        tipoAutenticacao,

      usuario:

        body.usuario?.trim() || null,

      headers,

      tentativas,

      timeout\_segundos:

        timeout,

      criado\_por: user.id,

    };

    /\*

     \* Senha:

     \*

     \* A coluna senha\_encriptada deve receber

     \* o valor tratado pelo backend.

     \*

     \* Por enquanto, somente grava quando

     \* uma senha foi informada.

     \*

     \* Se você já possui uma função de

     \* criptografia, devemos utilizá-la aqui.

     \*/

    if (body.senha) {

      insertData.senha\_encriptada =

        body.senha;

    }

    const { data, error } =

      await supabase

        .from("tasks")

        .insert(insertData)

        .select(TASK\_FIELDS)

        .single();

    if (error) {

      console.error(

        "[tarefas] Erro ao criar tarefa:",

        error

      );

      return jsonError(

        "Não foi possível criar a tarefa.",

        500

      );

    }

    console.log(

      \`[tarefas] Tarefa criada: ${data.id}\`

    );

    const avisoCronJob = await sincronizarCronJob(supabase, data);

    return NextResponse.json(

      {

        success: true,

        data,

        message:

          "Tarefa criada com sucesso.",

        aviso: avisoCronJob,

      },

      { status: 201 }

    );

  } catch (error) {

    console.error(

      "[tarefas] POST erro inesperado:",

      error

    );

    return jsonError(

      "Erro interno ao criar a tarefa.",

      500

    );

  }

}

/\*\*

 \* PATCH /api/tarefas

 \*

 \* Atualiza uma tarefa existente.

 \*/

export async function PATCH(

  request: NextRequest

) {

  try {

    const supabase = await createClient();

    const {

      data: { user },

      error: authError,

    } = await supabase.auth.getUser();

    if (authError || !user) {

      console.error(

        "[tarefas] Usuário não autenticado:",

        authError?.message

      );

      return jsonError(

        "Usuário não autenticado.",

        401

      );

    }

    const body =

      (await request.json()) as TaskPayload;

    if (!body.id) {

      return jsonError(

        "ID da tarefa não informado."

      );

    }

    const updateData: Record<

      string,

      unknown

    > = {};

    if (body.nome !== undefined) {

      const nome =

        body.nome.trim();

      if (!nome) {

        return jsonError(

          "O nome da tarefa é obrigatório."

        );

      }

      updateData.nome = nome;

    }

    if (body.descricao !== undefined) {

      updateData.descricao =

        body.descricao?.trim() ||

        null;

    }

    if (body.endpoint !== undefined) {

      const endpoint =

        body.endpoint.trim();

      if (!endpoint) {

        return jsonError(

          "O endpoint da tarefa é obrigatório."

        );

      }

      updateData.endpoint =

        endpoint;

    }

    if (body.metodo !== undefined) {

      const metodo =

        body.metodo

          .trim()

          .toUpperCase();

      const metodosPermitidos = [

        "GET",

        "POST",

        "PUT",

        "PATCH",

        "DELETE",

      ];

      if (

        !metodosPermitidos.includes(

          metodo

        )

      ) {

        return jsonError(

          "Método HTTP inválido."

        );

      }

      updateData.metodo =

        metodo;

    }

    if (

      body.cron\_expression !==

      undefined

    ) {

      const cronExpression =

        body.cron\_expression.trim();

      if (!cronExpression) {

        return jsonError(

          "A expressão de agendamento é obrigatória."

        );

      }

      updateData.cron\_expression =

        cronExpression;

    }

    if (

      body.ativa !== undefined

    ) {

      updateData.ativa =

        Boolean(body.ativa);

    }

    if (

      body.tipo\_agendamento !==

      undefined

    ) {

      updateData.tipo\_agendamento =

        body.tipo\_agendamento;

    }

    if (

      body.intervalo\_minutos !==

      undefined

    ) {

      updateData.intervalo\_minutos =

        body.intervalo\_minutos;

    }

    if (

      body.horario !== undefined

    ) {

      updateData.horario =

        body.horario || null;

    }

    if (

      body.data\_execucao !==

      undefined

    ) {

      updateData.data\_execucao =

        body.data\_execucao || null;

    }

    if (

      body.dia\_mes !== undefined

    ) {

      updateData.dia\_mes =

        body.dia\_mes;

    }

    if (body.mes !== undefined) {

      updateData.mes =

        body.mes;

    }

    if (body.ano !== undefined) {

      updateData.ano =

        body.ano;

    }

    if (

      body.timezone !== undefined

    ) {

      updateData.timezone =

        body.timezone ||

        "America/Sao\_Paulo";

    }

    if (

      body.tipo\_autenticacao !==

      undefined

    ) {

      updateData.tipo\_autenticacao =

        body.tipo\_autenticacao ===

          "nenhuma" ||

        !body.tipo\_autenticacao

          ? null

          : body.tipo\_autenticacao;

    }

    if (

      body.usuario !== undefined

    ) {

      updateData.usuario =

        body.usuario?.trim() ||

        null;

    }

    if (

      body.headers !== undefined

    ) {

      updateData.headers =

        Array.isArray(body.headers)

          ? body.headers.filter(

              (header) =>

                header &&

                header.key?.trim() &&

                header.value?.trim()

            )

          : [];

    }

    if (

      body.senha !== undefined &&

      body.senha !== null &&

      body.senha !== ""

    ) {

      updateData.senha\_encriptada =

        body.senha;

    }

    if (

      body.tentativas !== undefined

    ) {

      const tentativas =

        Number(body.tentativas);

      if (

        !Number.isInteger(

          tentativas

        ) ||

        tentativas < 1 ||

        tentativas > 20

      ) {

        return jsonError(

          "O número de tentativas deve estar entre 1 e 20."

        );

      }

      updateData.tentativas =

        tentativas;

    }

    if (

      body.timeout\_segundos !==

      undefined

    ) {

      const timeout =

        Number(

          body.timeout\_segundos

        );

      if (

        !Number.isInteger(

          timeout

        ) ||

        timeout < 1 ||

        timeout > 300

      ) {

        return jsonError(

          "O timeout deve estar entre 1 e 300 segundos."

        );

      }

      updateData.timeout\_segundos =

        timeout;

    }

    if (

      Object.keys(updateData)

        .length === 0

    ) {

      return jsonError(

        "Nenhum dado foi informado para atualização."

      );

    }

    const { data, error } =

      await supabase

        .from("tasks")

        .update(updateData)

        .eq("id", body.id)

        .select(TASK\_FIELDS)

        .single();

    if (error) {

      console.error(

        "[tarefas] Erro ao atualizar tarefa:",

        error

      );

      return jsonError(

        "Não foi possível atualizar a tarefa.",

        500

      );

    }

    console.log(

      \`[tarefas] Tarefa atualizada: ${body.id}\`

    );

    const CAMPOS\_QUE\_AFETAM\_O\_JOB = [

      "nome",

      "ativa",

      "tipo\_agendamento",

      "intervalo\_minutos",

      "horario",

      "data\_execucao",

      "cron\_expression",

      "timezone",

      "timeout\_segundos",

    ];

    const precisaSincronizar =

      CAMPOS\_QUE\_AFETAM\_O\_JOB.some((campo) => campo in updateData) ||

      !data.cronjob\_id;

    const avisoCronJob = precisaSincronizar

      ? await sincronizarCronJob(supabase, data)

      : null;

    return NextResponse.json({

      success: true,

      data,

      message:

        "Tarefa atualizada com sucesso.",

      aviso: avisoCronJob,

    });

  } catch (error) {

    console.error(

      "[tarefas] PATCH erro inesperado:",

      error

    );

    return jsonError(

      "Erro interno ao atualizar a tarefa.",

      500

    );

  }

}

/\*\*

 \* DELETE /api/tarefas?id=UUID

 \*

 \* Exclui uma tarefa.

 \*/

export async function DELETE(

  request: NextRequest

) {

  try {

    const supabase = await createClient();

    const {

      data: { user },

      error: authError,

    } = await supabase.auth.getUser();

    if (authError || !user) {

      console.error(

        "[tarefas] Usuário não autenticado:",

        authError?.message

      );

      return jsonError(

        "Usuário não autenticado.",

        401

      );

    }

    const { searchParams } =

      new URL(request.url);

    const id =

      searchParams.get("id");

    if (!id) {

      return jsonError(

        "ID da tarefa não informado."

      );

    }

    /\*

     \* Busca o cronjob\_id para excluir o job correspondente no

     \* cron-job.org. Falha ao excluir remotamente não impede a exclusão

     \* local — apenas fica logada (o job ficaria órfão no cron-job.org,

     \* chamando um endpoint que passa a responder 404).

     \*/

    const { data: taskExistente } = await supabase

      .from("tasks")

      .select("cronjob\_id")

      .eq("id", id)

      .maybeSingle();

    if (taskExistente?.cronjob\_id) {

      try {

        await excluirCronJob(taskExistente.cronjob\_id);

      } catch (cronJobError) {

        console.error(

          "[tarefas] Erro ao excluir job no cron-job.org:",

          cronJobError

        );

      }

    }

    /\*

     \* Remove primeiro o histórico

     \* de execuções.

     \*/

    const {

      error: executionsError,

    } = await supabase

      .from("task\_executions")

      .delete()

      .eq("task\_id", id);

    if (executionsError) {

      console.error(

        "[tarefas] Erro ao remover histórico:",

        executionsError

      );

      return jsonError(

        "Não foi possível remover o histórico da tarefa.",

        500

      );

    }

    const { error } =

      await supabase

        .from("tasks")

        .delete()

        .eq("id", id);

    if (error) {

      console.error(

        "[tarefas] Erro ao excluir tarefa:",

        error

      );

      return jsonError(

        "Não foi possível excluir a tarefa.",

        500

      );

    }

    console.log(

      \`[tarefas] Tarefa excluída: ${id}\`

    );

    return NextResponse.json({

      success: true,

      message:

        "Tarefa excluída com sucesso.",

    });

  } catch (error) {

    console.error(

      "[tarefas] DELETE erro inesperado:",

      error

    );

    return jsonError(

      "Erro interno ao excluir a tarefa.",

      500

    );

  }

}
